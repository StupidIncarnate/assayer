## Types

```ts
export interface PieceSlot {
  name: string;
  accepts: readonly PieceTag[];
  position: Position | 'inherit';
  segment: string;                          // folder segment when another piece is placed here
  form: 'hole' | 'statements' | 'params';   // a code parameter, a $stmts marker, or $params
  arm?: 'return' | 'log' | 'yield' | 'push' | 'throw'; // what an arm becomes in a statement slot of a container
  under?: string;                           // the arm of the piece whose code runs this slot
  marker?: ts.CallExpression;               // the $stmts or $expr call, for 'statements'
  hole?: { type: string; symbol: ts.Symbol }; // for 'hole'
}

export interface LoadedPiece {
  name: string;                             // file name without '.piece.ts' or '.shim.ts'
  origin: 'piece' | 'shim';
  description: string;
  folder: string;
  tags: readonly PieceTag[];
  slots: readonly PieceSlot[];              // sorted by name
  arms: Partial<Record<Position, readonly { name: string; cause?: string }[] | 'leaves'>>;
  leaf?: string;
  axes: readonly string[];                  // axis names, checked against axisStatics by the loader
  levelCap: number;
  defaultType?: string;                     // the type argument a specimen uses unless it varies the type axis
  reach?: string;
  returnType: string;
  typeParameter?: string;
  allowedTypeArguments: readonly string[];
  anchors: Record<string, unknown>;
  armMarkers: readonly string[];            // every $arm name in code, in source order
  binds: readonly string[];                 // names code declares (const, let, parameter), which a reader may read
  helpers: readonly ts.Statement[];         // statements of the helpers arrow, written at the top of the file
  form?: { kind: 'call' | 'getter' | 'method'; name: string };   // shims only
  builtin?: string;
  range?: { min: number; max: number; maxExclusive: boolean; whole: boolean };
  arrow: ts.ArrowFunction;
  sourceFile: ts.SourceFile;
}
```

`PieceTag` and `Position` come from the contracts of tasks 1.2.1 and 1.2.2, imported with `import type`. A
comment on `under` says: a slot with `under` runs only when its piece takes that arm, so the planner may nest
a piece there and the engine knows which arm's condition guards it.
