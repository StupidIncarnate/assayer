## Types

```ts
// One leaf: a value a hole reads. `name` is the identifier it renders as for a param, const or prop.
// `binding` is a read of a name a piece earlier in the same scope declared (`size` after `const size = count`).
export type PieceLeaf = { kind: 'leaf'; source: SourceKind | 'binding'; name: string; type: string; value?: unknown };
// A snippet is code no piece describes, from a composition card. It holds no branch.
export type PieceSnippet = { kind: 'snippet'; text: string; form: 'expression' | 'statement' };
// One node: a piece in a position, each slot filled by an ordered list of trees (a statement slot can hold
// several statements in a row; an expression slot holds exactly one).
export type PieceTree =
  | PieceLeaf
  | PieceSnippet
  | { kind: 'node'; piece: LoadedPiece; typeArgument?: string; position: Position; slots: Record<string, readonly PieceTree[]>; rename?: Record<string, string> };

export type PieceHelper =
  | { name: string; tree: PieceTree }
  | { name: string; typeAlias: string }
  | { name: string; enumMembers: readonly string[] };

export interface PiecePlan {
  folder: string;               // 'src/const/in/for-of-body/if-then'
  focus: string;                // the piece the specimen is about
  axis: AxisValue;              // the one thing it varies
  level: number;
  tree: PieceTree;              // the root is the container: 'module' or a function or class piece
  siblings: readonly { file: string; tree?: PieceTree; reexport?: { name: string; from: string } }[]; // file-boundary companions
  companion?: 'harness' | 'stub-agrees' | 'stub-missing-property' | 'stub-contradicts';
  stubCorrection?: { type: string; property: string; values: readonly unknown[] };
  exportName?: string | null;   // a card's own export name (task 1.3.4); absent means `entry` or `Entry`
  compilerOptions?: Record<string, boolean>;               // the compiler-options axis
  callSequence?: number;        // the most calls one case makes in a row
  sharedState?: 'module-let' | 'this-field' | 'closure-capture' | 'opaque-write';
  helpers: readonly PieceHelper[];
  card: boolean;
}
```

`PieceTree` replaces `FillTree` and `PiecePlan` replaces `SpecimenPlan` when task 9.7.2 deletes the old
contracts. Both new files import `LoadedPiece` (task 1.3.5), `Position` (task 1.2.2), `SourceKind` (task 1.2.3)
and `AxisValue` (task 1.2.4) with `import type`.
